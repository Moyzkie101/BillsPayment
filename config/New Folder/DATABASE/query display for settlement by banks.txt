WITH partner_name_list AS (
    SELECT 
        mpm.partner_id,
        mpm.partner_id_kpx,
        mpm.partner_name,
        mpm.partner_accName,
        mpm.bank_accNumber,
        mpm.bank AS bank_name,
        mpm.settled_online_check,
        mpm.charge_to,
        mpm.charge_sched
    FROM masterdata.partner_masterfile mpm
    WHERE mpm.status = 'ACTIVE'
),

transaction_data AS (
    SELECT 
        mbt.partner_id,
        mbt.partner_id_kpx,
        COUNT(*) AS volume_count,
        SUM(mbt.amount_paid) AS principal_amount_paid,
        SUM(mbt.charge_to_customer + mbt.charge_to_partner) AS charges
    FROM mldb.billspayment_transaction mbt
    WHERE mbt.status IS NULL
      AND mbt.branch_id NOT IN ('1','2','4937','4938','4962','4987','4993','4944')
      AND DATE(mbt.datetime) = '2026-01-05'
    GROUP BY mbt.partner_id, mbt.partner_id_kpx
),

principal_adjustment_data AS (
    SELECT 
		mbt.partner_id,
		mbt.partner_id_kpx,
		SUM(mbt.amount_paid) AS principal_adjustment
	FROM mldb.billspayment_transaction mbt
	WHERE mbt.status IS NULL
	  AND mbt.branch_id NOT IN ('1','2','4937','4938','4962','4987','4993','4944')
	AND DATE(mbt.settlement_date) = '2026-01-05'
	GROUP BY 
		mbt.partner_id,
		mbt.partner_id_kpx
)

SELECT
    pml.partner_name,
    pml.partner_accName,
    pml.bank_accNumber,
    pml.bank_name,
    pml.settled_online_check,
    pml.charge_to,
    pml.charge_sched,
    COALESCE(td.volume_count, 0) AS volume_count,
    COALESCE(td.principal_amount_paid, 0) AS principal_amount_paid,
    COALESCE(td.charges, 0) AS charges,
    COALESCE(pad.principal_adjustment-10, 0) AS principal_adjustment,
    COALESCE((td.principal_amount_paid - td.charges) + (pad.principal_adjustment-10),0) AS amount_for_settlement
FROM partner_name_list pml
LEFT JOIN transaction_data td
    ON (td.partner_id = pml.partner_id OR td.partner_id_kpx = pml.partner_id_kpx)
LEFT JOIN principal_adjustment_data pad
    ON (pad.partner_id = pml.partner_id OR pad.partner_id_kpx = pml.partner_id_kpx);
